# Complete source includes reviewed original-expression programs and text goldens.
set(NOBIHAZA_LOCAL_CONTENT_INPUTS "" CACHE FILEPATH "Lawful local manifest for seven fixed content includes")
set(NOBIHAZA_LOCAL_TEXT_FIXTURES "${CMAKE_SOURCE_DIR}/tests/fixtures/original-text.json" CACHE FILEPATH "Lawful local original-text test goldens")
if(EXISTS "${NOBIHAZA_LOCAL_TEXT_FIXTURES}")
    file(READ "${CMAKE_SOURCE_DIR}/tools/public-compile-contract.json" NOBIHAZA_COMPILE_CONTRACT)
    string(JSON NOBIHAZA_TEXT_SHA GET "${NOBIHAZA_COMPILE_CONTRACT}" text_fixture_sha256)
    file(SHA256 "${NOBIHAZA_LOCAL_TEXT_FIXTURES}" NOBIHAZA_LOCAL_TEXT_SHA)
    if(NOT NOBIHAZA_LOCAL_TEXT_SHA STREQUAL NOBIHAZA_TEXT_SHA)
        message(FATAL_ERROR "Changed original-text test goldens; review the local fixture contract")
    endif()
endif()
set(NOBIHAZA_CONTENT_INCLUDES
    src/game/dekisugi_program.inc src/game/dora_program.inc src/game/cash_program.inc
    src/game/warehouse_g4_program.inc src/game/final_boss_program.inc
    tools/g3-route-doors.inc tools/g3-access-cold-room.inc)
if(NOBIHAZA_LOCAL_CONTENT_INPUTS)
    if(NOT EXISTS "${NOBIHAZA_LOCAL_CONTENT_INPUTS}")
        message(FATAL_ERROR "NOBIHAZA_LOCAL_CONTENT_INPUTS must name an existing lawful local manifest")
    endif()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(NOBIHAZA_GENERATED_CONTENT "${CMAKE_BINARY_DIR}/local-content-includes")
    set(NOBIHAZA_GENERATED_FILES)
    foreach(NOBIHAZA_INCLUDE IN LISTS NOBIHAZA_CONTENT_INCLUDES)
        list(APPEND NOBIHAZA_GENERATED_FILES "${NOBIHAZA_GENERATED_CONTENT}/${NOBIHAZA_INCLUDE}")
    endforeach()
    add_custom_target(nobi_local_compile_content
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/public_source_lower.py"
            --inputs "${NOBIHAZA_LOCAL_CONTENT_INPUTS}" --output "${NOBIHAZA_GENERATED_CONTENT}"
        BYPRODUCTS ${NOBIHAZA_GENERATED_FILES}
        VERBATIM)
else()
    foreach(NOBIHAZA_INCLUDE IN LISTS NOBIHAZA_CONTENT_INCLUDES)
        if(NOT EXISTS "${CMAKE_SOURCE_DIR}/${NOBIHAZA_INCLUDE}")
            message(FATAL_ERROR "Required content include is missing. Restore the complete source checkout, or supply NOBIHAZA_LOCAL_CONTENT_INPUTS.")
        endif()
    endforeach()
endif()
